// --- Utility: Color conversions & ΔE2000
function hexToRgb(hex) {
  const m = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
  return m ? { r: parseInt(m[1], 16) / 255, g: parseInt(m[2], 16) / 255, b: parseInt(m[3], 16) / 255 } : null;
}
function invGamma(u) {
  return u <= 0.04045 ? u / 12.92 : Math.pow((u + 0.055) / 1.055, 2.4);
}
function rgbToXyz({ r, g, b }) {
  const R = invGamma(r), G = invGamma(g), B = invGamma(b);
  const x = R * 0.4124564 + G * 0.3575761 + B * 0.1804375;
  const y = R * 0.2126729 + G * 0.7151522 + B * 0.0721750;
  const z = R * 0.0193339 + G * 0.1191920 + B * 0.9503041;
  return { x, y, z };
}
function fLab(t) { return t > 216 / 24389 ? Math.cbrt(t) : (24389 / 27) * t + 16 / 116; }
function xyzToLab({ x, y, z }) {
  // D65 reference white
  const Xn = 0.95047, Yn = 1.00000, Zn = 1.08883;
  const fx = fLab(x / Xn), fy = fLab(y / Yn), fz = fLab(z / Zn);
  const L = 116 * fy - 16;
  const a = 500 * (fx - fy);
  const b = 200 * (fy - fz);
  return { L, a, b };
}
function hexToLab(hex) { return xyzToLab(rgbToXyz(hexToRgb(hex))); }
function deg2rad(d) { return (d * Math.PI) / 180; }
function rad2deg(r) { return (r * 180) / Math.PI; }
// CIEDE2000 implementation
function deltaE2000(lab1, lab2) {
  const { L: L1, a: a1, b: b1 } = lab1;
  const { L: L2, a: a2, b: b2 } = lab2;
  const kL = 1, kC = 1, kH = 1;
  const C1 = Math.hypot(a1, b1);
  const C2 = Math.hypot(a2, b2);
  const Cbar = (C1 + C2) / 2;
  const G = 0.5 * (1 - Math.sqrt(Math.pow(Cbar, 7) / (Math.pow(Cbar, 7) + Math.pow(25, 7))));
  const a1p = (1 + G) * a1;
  const a2p = (1 + G) * a2;
  const C1p = Math.hypot(a1p, b1);
  const C2p = Math.hypot(a2p, b2);
  const h1p = (Math.atan2(b1, a1p) + 2 * Math.PI) % (2 * Math.PI);
  const h2p = (Math.atan2(b2, a2p) + 2 * Math.PI) % (2 * Math.PI);
  const dLp = L2 - L1;
  const dCp = C2p - C1p;
  let dhp = h2p - h1p;
  if (dhp > Math.PI) dhp -= 2 * Math.PI;
  if (dhp < -Math.PI) dhp += 2 * Math.PI;
  const dHp = 2 * Math.sqrt(C1p * C2p) * Math.sin(dhp / 2);
  const Lbarp = (L1 + L2) / 2;
  const Cbarp = (C1p + C2p) / 2;
  let hbarp = h1p + h2p;
  if (Math.abs(h1p - h2p) > Math.PI) hbarp += 2 * Math.PI;
  hbarp /= 2;
  const T = 1 - 0.17 * Math.cos(hbarp - deg2rad(30)) + 0.24 * Math.cos(2 * hbarp) + 0.32 * Math.cos(3 * hbarp + deg2rad(6)) - 0.20 * Math.cos(4 * hbarp - deg2rad(63));
  let dRo = deg2rad(30) * Math.exp(-Math.pow((rad2deg(hbarp) - 275) / 25, 2));
  const Rc = 2 * Math.sqrt(Math.pow(Cbarp, 7) / (Math.pow(Cbarp, 7) + Math.pow(25, 7)));
  const Sl = 1 + (0.015 * Math.pow(Lbarp - 50, 2)) / Math.sqrt(20 + Math.pow(Lbarp - 50, 2));
  const Sc = 1 + 0.045 * Cbarp;
  const Sh = 1 + 0.015 * Cbarp * T;
  const Rt = -Math.sin(2 * dRo) * Rc;
  return Math.sqrt(
    Math.pow(dLp / (kL * Sl), 2) +
    Math.pow(dCp / (kC * Sc), 2) +
    Math.pow(dHp / (kH * Sh), 2) +
    Rt * (dCp / (kC * Sc)) * (dHp / (kH * Sh))
  );
}

// --- Components -------------------------------------------------------
function Section({ title, children, subtitle }) {
  const exportPNG = async () => {
    try {
      setExporting(true);
      const node = captureRef.current;
      if (!node) return;
      const canvas = await html2canvas(node, { backgroundColor: null, scale: 2 });
      const url = canvas.toDataURL("image/png");
      const a = document.createElement("a");
      a.href = url; a.download = `GemILike_${Date.now()}.png`; a.click();
    } catch (e) {
      console.error(e);
    } finally {
      setExporting(false);
    }
  };

  const exportJSONServer = async (data) => {
    try {
      const res = await fetch('/api/export', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
      const blob = await res.blob();
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url; a.download = `${data.id || 'gemilike'}.json`; a.click();
      URL.revokeObjectURL(url);
    } catch (e) {
      console.error(e);
    }
  };

  return (
    <section className="w-full max-w-6xl mx-auto px-4 sm:px-6 lg:px-8 py-8">
      <div className="mb-6">
        <h2 className="text-2xl sm:text-3xl font-semibold tracking-tight text-[#2E2A47] dark:text-violet-100">{title}</h2>
        {subtitle && <p className="text-sm text-slate-600/80 mt-1">{subtitle}</p>}
      </div>
      {children}
    </section>
  );
}

function GradientBar({ colors }) {
  const canvasRef = useRef(null);
  React.useEffect(() => {
    const c = canvasRef.current;
    if (!c) return;
    const ctx = c.getContext("2d");
    const g = ctx.createLinearGradient(0, 0, c.width, 0);
    const n = colors.length;
    colors.forEach((col, i) => g.addColorStop(i / (n - 1), col));
    ctx.fillStyle = g;
    ctx.fillRect(0, 0, c.width, c.height);
  }, [colors]);
  return <canvas ref={canvasRef} className="w-full h-6 rounded-xl" width={800} height={40} />;
}

function downloadText(filename, text) {

  URL.revokeObjectURL(url);
}

// ========================= Upload Panel ===============================
function UploadPanel({ onImport }) {
  const inputRef = useRef(null);
  const handleFile = (f) => {
    const reader = new FileReader();
    reader.onload = () => {
      try {
        const data = JSON.parse(String(reader.result));
        onImport?.(data);
      } catch (e) { alert('Ungültiges JSON.'); }
    };
    reader.readAsText(f);
  };
  return (
    <div className="rounded-3xl border border-white/10 bg-white/5 p-5 text-sm">
      <p className="text-slate-300">Format: Array von Objekten {`{ id, name, gia, gradient[], pleochro[], light }`}</p>
      <div className="mt-3 flex gap-3 items-center">
        <button onClick={() => inputRef.current?.click()} className="rounded-2xl bg-white/10 px-4 py-2 hover:bg-white/20">JSON auswählen…</button>
        <input ref={inputRef} type="file" accept="application/json" className="hidden" onChange={(e) => e.target.files?.[0] && handleFile(e.target.files[0])} />
        <button onClick={() => {
          const sample = [
            { id: 'demo-stone', name: 'Demo Stone', origin: 'N/A', gia: { hue: 'B', tone: '5', sat: '3' }, note: 'Demo', gradient: ['#dde4ff','#9fb2ff','#6f85f0','#4c63cc','#3146a1'], pleochro: ['#9fb2ff','#6f85f0','#3146a1'], light: 'D55' }
          ];
          onImport?.(sample);
        }} className="rounded-2xl border border-white/20 px-4 py-2 hover:bg-white/10">Demo laden</button>
      </div>
    </div>
  );
}
